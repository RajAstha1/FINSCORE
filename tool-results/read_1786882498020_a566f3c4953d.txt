   258→   258→function formatCurrency(amount: number): string {
   259→   259→  if (amount >= 10000000) {
   260→   260→    return `₹${(amount / 10000000).toFixed(2)} Cr`;
   261→   261→  }
   262→   262→  if (amount >= 100000) {
   263→   263→    return `₹${(amount / 100000).toFixed(2)} L`;
   264→   264→  }
   265→   265→  return new Intl.NumberFormat('en-IN', {
   266→   266→    style: 'currency',
   267→   267→    currency: 'INR',
   268→   268→    maximumFractionDigits: 0,
   269→   269→  }).format(amount);
   270→   270→}
   271→   271→
   272→   272→function scoreColor(score: number): string {
   273→   273→  if (score >= 75) return 'text-emerald-600 dark:text-emerald-400';
   274→   274→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   275→   275→  if (score >= 45) return 'text-orange-600 dark:text-orange-400';
   276→   276→  return 'text-red-600 dark:text-red-400';
   277→   277→}
   278→   278→
   279→   279→function scoreBg(score: number): string {
   280→   280→  if (score >= 75) return 'bg-emerald-50 dark:bg-emerald-950/30';
   281→   281→  if (score >= 60) return 'bg-amber-50 dark:bg-amber-950/30';
   282→   282→  if (score >= 45) return 'bg-orange-50 dark:bg-orange-950/30';
   283→   283→  return 'bg-red-50 dark:bg-red-950/30';
