   812→export function ApplicationWizard() {
   813→  const [currentStep, setCurrentStep] = useState(1);
   814→  const [direction, setDirection] = useState<'forward' | 'backward'>('forward');
   815→  const [isSubmitting, setIsSubmitting] = useState(false);
   816→  const [submitError, setSubmitError] = useState<string | null>(null);
   817→  const [successResult, setSuccessResult] = useState<{
   818→    score: { totalScore: number; riskGrade: string; decisionType: string };
   819→    appNumber: string;
   820→  } | null>(null);
   821→
   822→  const token = useAuthStore((s) => s.token);
   823→  const navigate = useAppStore((s) => s.navigate);
   824→
   825→  const form = useForm<ApplicationFormData>({
   826→    resolver: zodResolver(fullSchema),
   827→    defaultValues: {
   828→      name: '',
   829→      aadhaarNumber: '',
   830→      panNumber: '',
   831→      dateOfBirth: '',
   832→      gender: '',
   833→      category: '',
   834→      educationLevel: '',
   835→      maritalStatus: '',
   836→      phone: '',
   837→      email: '',
   838→      address: '',
   839→      state: '',
   840→      district: '',
   841→      pincode: '',
   842→      occupation: '',
   843→      monthlyIncome: undefined,
   844→      bankName: '',
   845→      bankAccount: '',
   846→      bankIfsc: '',
   847→      loanAmount: undefined,
   848→      loanPurpose: '',
   849→      loanTenure: undefined,
   850→      schemeType: '',
   851→      electricityConsistency: 50,
   852→      mobileRechargeConsistency: 50,
   853→      utilityPaymentHistory: 50,
   854→      repaymentHistory: 50,
   855→      consentCreditBureau: false,
   856→      consentDataSharing: false,
   857→      consentAccuracy: false,
   858→    },
   859→    mode: 'onBlur',
   860→  });
   861→
   862→  const validateCurrentStep = useCallback(
   863→    async (step: number) => {
   864→      const fields = STEP_FIELDS[step];
   865→      if (!fields) return true;
   866→      return form.trigger(fields as string[]);
   867→    },
   868→    [form],
   869→  );
   870→
   871→  const goNext = useCallback(async () => {
   872→    const valid = await validateCurrentStep(currentStep);
   873→    if (valid) {
   874→      setDirection('forward');
   875→      setCurrentStep((s) => Math.min(s + 1, 6));
   876→    }
   877→  }, [currentStep, validateCurrentStep]);
   878→
   879→  const goBack = useCallback(() => {
   880→    setDirection('backward');
   881→    setCurrentStep((s) => Math.max(s - 1, 1));
   882→  }, []);
   883→
   884→  const handleSubmit = useCallback(async () => {
   885→    // Validate the last step (consent)
   886→    const valid = await validateCurrentStep(6);
   887→    if (!valid) return;
   888→
   889→    setIsSubmitting(true);
   890→    setSubmitError(null);
   891→
   892→    try {
   893→      const values = form.getValues();
   894→      const res = await fetch('/api/applications', {
   895→        method: 'POST',
   896→        headers: {
   897→          'Content-Type': 'application/json',
   898→          Authorization: `Bearer ${token}`,
   899→        },
   900→        body: JSON.stringify(values),
   901→      });
   902→
   903→      if (!res.ok) {
   904→        const errData = await res.json().catch(() => ({ error: 'Submission failed' }));
   905→        throw new Error(errData.error || `HTTP ${res.status}`);
   906→      }
   907→
   908→      const responseData = await res.json();
   909→      setSuccessResult({
   910→        score: responseData.score,
   911→        appNumber: responseData.application.applicationNumber,
   912→      });
   913→
   914→      // Navigate after 2 seconds
   915→      setTimeout(() => {
   916→        navigate('application-detail', { id: responseData.application.id });
   917→      }, 2000);
   918→    } catch (err) {
   919→      setSubmitError(err instanceof Error ? err.message : 'Something went wrong');
   920→    } finally {
   921→      setIsSubmitting(false);
   922→    }
   923→  }, [form, token, validateCurrentStep, navigate]);
   924→
   925→  const progressValue = ((currentStep - 1) / 5) * 100;
   926→
   927→  const stepVariants = {
   928→    enter: (dir: 'forward' | 'backward') => ({
   929→      x: dir === 'forward' ? 300 : -300,
   930→      opacity: 0,
   931→    }),
   932→    center: {
   933→      x: 0,
   934→      opacity: 1,
   935→    },
   936→    exit: (dir: 'forward' | 'backward') => ({
   937→      x: dir === 'forward' ? -300 : 300,
   938→      opacity: 0,
   939→    }),
   940→  };
   941→
   942→  return (
   943→    <div className="mx-auto w-full max-w-4xl space-y-6">
   944→      {/* Header */}
   945→      <div>
   946→        <h1 className="text-2xl font-bold tracking-tight">New Loan Application</h1>

   948→      </div>
   949→
   950→      {/* Progress Bar */}
   951→      <Card className="overflow-hidden">

   961→                  <div
   962→                    className={`flex size-8 items-center justify-center rounded-full border-2 transition-colors ${
   963→                      isCompleted
   964→                        ? 'border-teal-500 bg-teal-500 text-white'
   965→                        : isCurrent
   966→                          ? 'border-teal-500 bg-white text-teal-600'
   967→                          : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   968→                    }`}

   970→                    {isCompleted ? (
   971→                      <CheckCircle2 className="size-4" />
   972→                    ) : (
   973→                      <StepIcon className="size-4" />
   974→                    )}
   975→                  </div>
   976→                  <span

   987→          <Progress value={progressValue} className="h-2" />
   988→          <p className="mt-1.5 text-center text-xs text-muted-foreground">
   989→            Step {currentStep} of 6 — {STEPS[currentStep - 1].name}
   990→          </p>
   991→        </CardContent>

   994→      {/* Step Card */}

  1008→            <motion.div
  1009→              key={currentStep}
  1010→              custom={direction}
  1011→              variants={stepVariants}
  1012→              initial="enter"
  1013→              animate="center"
