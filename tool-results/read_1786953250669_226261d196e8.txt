   211→/** Animated SVG circular gauge */
   212→function ScoreGauge({ score, size = 160, strokeWidth = 12, color = '#14b8a6' }: { score: number; size?: number; strokeWidth?: number; color?: string }) {
   213→  const radius = (size - strokeWidth) / 2;
   214→  const circumference = 2 * Math.PI * radius;
   215→  const motionVal = useMotionValue(0);
   216→  const strokeDashoffset = useTransform(motionVal, (v) => circumference - (v / 100) * circumference);
   217→
   218→  useEffect(() => {
   219→    const controls = animate(motionVal, score, { duration: 1.5, ease: 'easeOut' });
   220→    return () => controls.stop();
   221→  }, [score, motionVal]);
   222→
   223→  return (
   224→    <div className="relative inline-flex items-center justify-center">
   225→      <svg width={size} height={size} className="-rotate-90">
   226→        <circle
   227→          cx={size / 2}
   228→          cy={size / 2}
   229→          r={radius}
   230→          fill="none"
   231→          stroke="currentColor"
   232→          strokeWidth={strokeWidth}
   233→          className="text-muted/30"
   234→        />
   235→        <motion.circle
   236→          cx={size / 2}
   237→          cy={size / 2}
   238→          r={radius}
   239→          fill="none"
   240→          stroke={color}
   241→          strokeWidth={strokeWidth}
   242→          strokeLinecap="round"
   243→          strokeDasharray={circumference}
   244→          style={{ strokeDashoffset }}
   245→        />
   246→      </svg>
   247→      <span className="absolute font-mono text-4xl font-bold text-foreground">
   248→        <motion.span>{useTransform(motionVal, (v) => Math.round(v))}</motion.span>
   249→      </span>
   250→    </div>
   251→  );
   252→}

   254→/** Mini circular gauge for sub-scores */
   255→function MiniGauge({ score, label, size = 72, strokeWidth = 6 }: { score: number; label: string; size?: number; strokeWidth?: number }) {
   256→  const radius = (size - strokeWidth) / 2;
   257→  const circumference = 2 * Math.PI * radius;
   258→  const pct = Math.min(100, Math.max(0, score));
   259→  const offset = circumference - (pct / 100) * circumference;
   260→  const color = score > 70 ? '#14b8a6' : score >= 50 ? '#f59e0b' : '#ef4444';
   261→
   262→  return (
   263→    <div className="flex flex-col items-center gap-1.5">
   264→      <div className="relative inline-flex items-center justify-center">
   265→        <svg width={size} height={size} className="-rotate-90">
   266→          <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />
   267→          <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth} strokeLinecap="round" strokeDasharray={circumference} strokeDashoffset={offset} style={{ transition: 'stroke-dashoffset 1s ease-out' }} />
   268→        </svg>
   269→        <span className="absolute font-mono text-sm font-bold text-foreground">{score}</span>
   270→      </div>
   271→      <span className="text-xs text-muted-foreground font-medium">{label}</span>
   272→    </div>
   273→  );
   274→}

   276→/** SHAP waterfall / contribution chart */
   277→function ShapChart({ shapValues }: { shapValues: ShapValue[] }) {
   278→  const sorted = [...shapValues].sort((a, b) => Math.abs(b.shapValue) - Math.abs(a.shapValue));
   279→  const maxAbs = Math.max(...sorted.map((s) => Math.abs(s.shapValue)), 1);
   280→
   281→  return (
   282→    <div className="space-y-2">
   283→      {/* Base value indicator */}
   284→      <div className="flex items-center gap-2 mb-3">
   285→        <div className="h-px flex-1 bg-border" />\n        <span className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">Base Value</span>
   286→        <div className="h-px flex-1 bg-border" />
   287→      </div>
   288→
   289→      {sorted.map((sv, i) => {
   290→        const widthPct = (Math.abs(sv.shapValue) / maxAbs) * 100;
   291→        const isPositive = sv.direction === 'positive';
   292→
   293→        return (
   294→          <motion.div
   295→            key={sv.feature}
   296→            initial={{ opacity: 0, x: isPositive ? 20 : -20 }}
   297→            animate={{ opacity: 1, x: 0 }}
   298→            transition={{ delay: i * 0.05, duration: 0.3 }}
   299→            className="group"
   300→          >
   301→            <div className="flex items-center gap-2 text-xs">
   302→              {/* Feature label */}
   303→              <span className="w-36 shrink-0 truncate text-right text-muted-foreground group-hover:text-foreground transition-colors">
   304→                {sv.feature}
   305→              </span>
   306→
   307→              {/* Bar area */}
   308→              <div className="relative flex-1 h-6 flex items-center">
   309→                {/* Base line */}
   310→                <div className="absolute left-1/2 -translate-x-px top-0 h-full w-px bg-border" />
   311→
   312→                {/* Bar */}
   313→                <motion.div
   314→                  initial={{ width: 0 }}
   315→                  animate={{ width: `${widthPct * 0.45}%` }}
   316→                  transition={{ delay: i * 0.05 + 0.15, duration: 0.5, ease: 'easeOut' }}
   317→                  className={`h-3.5 rounded-sm ${
   318→                    isPositive
   319→                      ? 'ml-[50%] bg-gradient-to-r from-teal-500/80 to-teal-400/60'
   320→                      : 'mr-[50%] bg-gradient-to-l from-red-500/80 to-red-400/60'
   321→                  }`}
   322→                  style={isPositive ? {} : { marginLeft: 'auto' }}
   323→                />
   324→              </div>
   325→
   326→              {/* Value */}
   327→              <span
   328→                className={`w-12 text-right font-mono font-medium shrink-0 ${
   329→                  isPositive ? 'text-teal-600 dark:text-teal-400' : 'text-red-600 dark:text-red-400'
   330→                }`}
   331→              >
   332→                {isPositive ? '+' : ''}
   333→                {sv.shapValue.toFixed(1)}
   334→              </span>
   335→            </div>
   336→          </motion.div>
   337→        );
   338→      })}
   339→
   340→      {/* Legend */}
   341→      <div className="flex items-center justify-center gap-4 pt-3">
   342→        <div className="flex items-center gap-1.5">
   343→          <div className="h-2.5 w-4 rounded-sm bg-gradient-to-r from-teal-500/80 to-teal-400/60" />\n          <span className="text-[10px] text-muted-foreground">Positive contribution</span>
   344→        </div>
   345→        <div className="flex items-center gap-1.5">
   346→          <div className="h-2.5 w-4 rounded-sm bg-gradient-to-l from-red-500/80 to-red-400/60" />
   347→          <span className="text-[10px] text-muted-foreground">Negative contribution</span>
   348→        </div>
   349→      </div>
   350→    </div>
   351→  );
   352→}

   364→  const [localStatus, setLocalStatus] = useState<string | null>(null);
   365→
   366→  /* ---- Data fetching ---- */
   367→  const { data: application, isLoading } = useQuery<Application>({
   368→    queryKey: ['application', id],
   369→    queryFn: async () => {
   370→      const res = await fetch(`/api/applications?search=${encodeURIComponent(id)}`, {
   371→        headers: { Authorization: `Bearer ${token}` },
   372→      });
   373→      if (!res.ok) throw new Error('Failed to fetch application');
