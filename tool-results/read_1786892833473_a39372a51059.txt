   150→function maskAadhaar(aadhaar?: string) {
   151→  if (!aadhaar || aadhaar.length < 4) return '••••••••••••';
   152→  return `${aadhaar.slice(0, 4)}••••••••${aadhaar.slice(-4)}`;
   153→}
   154→
   155→function formatDate(dateStr?: string | null) {
   156→  if (!dateStr) return '—';
   157→  return new Date(dateStr).toLocaleDateString('en-IN', {
   158→    day: '2-digit', month: 'short', year: 'numeric',
   159→  });
   160→}
   161→
   162→function formatDateTime(dateStr?: string | null) {
   163→  if (!dateStr) return '—';
   164→  return new Date(dateStr).toLocaleString('en-IN', {
   165→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   166→  });
   167→}
   168→
   169→function formatCurrency(amount: number) {
   170→  return `\u20B9${amount.toLocaleString('en-IN')}`;
   171→}
   172→
   173→function InfoRow({ label, value, icon: Icon }: { label: string; value: string | number | null | undefined; icon?: React.ElementType }) {
   174→  return (
   175→    <div className="flex items-center justify-between py-1.5">
   176→      <span className="text-sm text-muted-foreground flex items-center gap-1.5">
   177→        {Icon && <Icon className="size-3.5" />}
   178→        {label}
   179→      </span>
   180→      <span className="text-sm font-medium text-right capitalize font-mono">{value ?? '—'}</span>
   181→    </div>
   182→  );
   183→}
   184→
   185→function formatDocType(type: string) {
   186→  return type.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   187→}
