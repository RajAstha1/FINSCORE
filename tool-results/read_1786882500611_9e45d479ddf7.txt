   257→
   258→function formatCurrency(amount: number): string {
   259→  if (amount >= 10000000) {
   260→    return `₹${(amount / 10000000).toFixed(2)} Cr`;
   261→  }
   262→  if (amount >= 100000) {
   263→    return `₹${(amount / 100000).toFixed(2)} L`;
   264→  }
   265→  return new Intl.NumberFormat('en-IN', {
   266→    style: 'currency',
   267→    currency: 'INR',
   268→    maximumFractionDigits: 0,
   269→  }).format(amount);
   270→}
   271→
   272→function scoreColor(score: number): string {
   273→  if (score >= 75) return 'text-emerald-600 dark:text-emerald-400';
   274→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   275→  if (score >= 45) return 'text-orange-600 dark:text-orange-400';
   276→  return 'text-red-600 dark:text-red-400';
   277→}
   278→
   279→function scoreBg(score: number): string {
   280→  if (score >= 75) return 'bg-emerald-50 dark:bg-emerald-950/30';
   281→  if (score >= 60) return 'bg-amber-50 dark:bg-amber-950/30';
   282→  if (score >= 45) return 'bg-orange-50 dark:bg-orange-950/30';
   283→  return 'bg-red-50 dark:bg-red-950/30';
   284→}

   386→/* ------------------------------------------------------------------ */
   387→/*  Custom Recharts Tooltip                                             */
   388→/* ------------------------------------------------------------------ */
   389→
   390→function CustomAreaTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   391→  if (!active || !payload || !payload.length) return null;
   392→  return (
   393→    <div className="rounded-lg border border-border bg-card p-3 shadow-lg">
   394→      <p className="text-sm font-medium mb-1.5">{label}</p>
   395→      {payload.map((item) => (
   396→        <div key={item.name} className="flex items-center gap-2 text-xs">
   397→          <span
   398→            className="w-2.5 h-2.5 rounded-full"
   399→            style={{ backgroundColor: item.color }}
   400→          />
   401→          <span className="text-muted-foreground capitalize">
   402→            {item.name.replace('_', ' ')}:
   403→          </span>
   404→          <span className="font-mono font-medium">{item.value}</span>
   405→        </div>
   406→      ))}
   407→    </div>
   408→  );
   409→}
