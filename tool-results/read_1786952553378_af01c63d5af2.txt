   890→   890→  const handleSubmit = useCallback(async () => {
   891→   891→    // Validate the last step (consent)
   892→   892→    const valid = await validateCurrentStep(6);
   893→   893→    if (!valid) return;
   894→   894→
   895→   895→    setIsSubmitting(true);
   896→   896→    setSubmitError(null);
   897→   897→
   898→   898→    try {
   899→   899→      const values = form.getValues();
   900→   900→      const res = await fetch('/api/applications', {
   901→   901→        method: 'POST',
   902→   902→        headers: {
   903→   903→          'Content-Type': 'application/json',
   904→   904→          Authorization: `Bearer ${token}`,
   905→   905→        },
   906→   906→        body: JSON.stringify(values),
   907→   907→      });
   908→   908→
   909→   909→      if (!res.ok) {
   910→   910→        const errData = await res.json().catch(() => ({ error: 'Submission failed' }));
   911→   911→        throw new Error(errData.error || `HTTP ${res.status}`);
   912→   912→      }
   913→   913→
   914→   914→      const responseData = await res.json();
   915→   915→      setSuccessResult({
   916→   916→        score: responseData.score,
   917→   917→        appNumber: responseData.application.applicationNumber,
   918→   918→      });
   919→   919→
   920→   920→      // Navigate after 2 seconds
   921→   921→      setTimeout(() => {
   922→   922→        navigate('application-detail', { id: responseData.application.id });
   923→   923→      }, 2000);
   924→   924→    } catch (err) {
   925→   925→      setSubmitError(err instanceof Error ? err.message : 'Something went wrong');
   926→   926→    } finally {
   927→   927→      setIsSubmitting(false);
   928→   928→    }
   929→   929→  }, [form, token, validateCurrentStep, navigate]);
   930→   930→
   931→   931→  const progressValue = ((currentStep - 1) / 5) * 100;
   932→   932→
   933→   933→  const stepVariants = {
   934→   934→    enter: (dir: 'forward' | 'backward') => ({
   935→   935→      x: dir === 'forward' ? 300 : -300,
   936→   936→      opacity: 0,
   937→   937→    }),
   938→   938→    center: {
   939→   939→      x: 0,
   940→   940→      opacity: 1,
   941→   941→    },
   942→   942→    exit: (dir: 'forward' | 'backward') => ({
   943→   943→      x: dir === 'forward' ? -300 : 300,
   944→   944→      opacity: 0,
   945→   945→    }),
   946→   946→  };
   947→   947→
   948→   948→  return (
   949→   949→    <div className="mx-auto w-full max-w-4xl space-y-6">
   950→   950→      {/* Header */}
   951→   951→      <div>
   952→   952→        <h1 className="text-2xl font-bold tracking-tight">New Loan Application</h1>
   953→   953→        <p className="text-sm text-muted-foreground">Complete all steps to submit</p>
   954→   954→      </div>
   955→   955→
   956→   956→      {/* Progress Bar */}
   957→   957→      <Card className="overflow-hidden">
   958→   958→        <CardContent className="p-4">
   959→   959→          {/* Step indicators */}
   960→   960→          <div className="mb-3 flex items-center justify-between">
   961→   961→            {STEPS.map((step) => {
   962→   962→              const StepIcon = step.icon;
   963→   963→              const isCompleted = step.id < currentStep;
   964→   964→              const isCurrent = step.id === currentStep;
   965→   965→              return (
   966→   966→                <div key={step.id} className="flex flex-1 flex-col items-center">
   967→   967→                  <div
   968→   968→                    className={`flex size-8 items-center justify-center rounded-full border-2 transition-colors ${
   969→   969→                      isCompleted
   970→   970→                        ? 'border-teal-500 bg-teal-500 text-white'
   971→   971→                        : isCurrent
   972→   972→                          ? 'border-teal-500 bg-white text-teal-600'
   973→   973→                          : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   974→   974→                    }`}
   975→   975→                  >
   976→   976→                    {isCompleted ? (
   977→   977→                      <CheckCircle2 className="size-4" />
   978→   978→                    ) : (
   979→   979→                      <StepIcon className="size-4" />
   980→   980→                    )}
   981→   981→                  </div>
   982→   982→                  <span
   983→   983→                    className={`mt-1 hidden text-[10px] font-medium sm:block ${
   984→   984→                      isCompleted || isCurrent ? 'text-teal-600' : 'text-muted-foreground'
   985→   985→                    }`}
   986→   986→                  >
   987→   987→                    {step.name}
   988→   988→                  </span>
   989→   989→                </div>
   990→   990→              );
   991→   991→            })}
   992→   992→          </div>
   993→   993→          <Progress value={progressValue} className="h-2" />
   994→   994→          <p className="mt-1.5 text-center text-xs text-muted-foreground">
   995→   995→            Step {currentStep} of 6 — {STEPS[currentStep - 1].name}
   996→   996→          </p>
   997→   997→        </CardContent>
   998→   998→      </Card>
   999→   999→
  1000→  1000→      {/* Step Card */}
  1001→  1001→      <Form {...form}>
  1002→  1002→      <Card>
  1003→  1003→        <CardHeader className="pb-2">
  1004→  1004→          <CardTitle className="flex items-center gap-2 text-base">
  1005→  1005→            {(() => {
  1006→  1006→              const StepIcon = STEPS[currentStep - 1].icon;
  1007→  1007→              return <StepIcon className="size-5 text-teal-600" />;
  1008→  1008→            })()}
  1009→  1009→            {STEPS[currentStep - 1].name}
  1010→  1010→          </CardTitle>
  1011→  1011→        </CardHeader>
  1012→  1012→        <CardContent className="pt-4">
  1013→  1013→          <AnimatePresence mode="wait" custom={direction}>
  1014→  1014→            <motion.div
  1015→  1015→              key={currentStep}
  1016→  1016→              custom={direction}
  1017→  1017→              variants={stepVariants}
  1018→  1018→              initial="enter"
  1019→  1019→              animate="center"
  1020→  1020→              exit="exit"
  1021→  1021→              transition={{ duration: 0.25, ease: 'easeInOut' }}
  1022→  1022→            >
  1023→  1023→              {currentStep === 1 && <PersonalInfoStep />}
  1024→  1024→              {currentStep === 2 && <AddressStep />}
  1025→  1025→              {currentStep === 3 && <IncomeStep />}
  1026→  1026→              {currentStep === 4 && <LoanDetailsStep />}
  1027→  1027→              {currentStep === 5 && <ConsumptionStep />}
  1028→  1028→              {currentStep === 6 && <ReviewStep />}
  1029→  1029→            </motion.div>
  1030→  1030→          </AnimatePresence>
  1031→  1031→
  1032→  1032→          {submitError && (
  1033→  1033→            <motion.div
  1034→  1034→              initial={{ opacity: 0, y: 5 }}
  1035→  1035→              animate={{ opacity: 1, y: 0 }}
  1036→  1036→              className="mt-4 rounded-lg border border-destructive/30 bg-destructive/10 p-3 text-sm text-destructive"
  1037→  1037→            >
  1038→  1038→              {submitError}
  1039→  1039→            </motion.div>
  1040→  1040→          )}
  1041→  1041→        </CardContent>
  1042→  1042→      </Card>
  1043→  1043→      </Form>
  1044→  1044→
  1045→  1045→      {/* Navigation Buttons */}
  1046→  1046→      <div className="flex items-center justify-between">
  1047→  1047→        <Button
  1048→  1048→          type="button"
  1049→  1049→          variant="outline"
  1050→  1050→          onClick={goBack}
  1051→  1051→          disabled={currentStep === 1 || isSubmitting}
  1052→  1052→        >
  1053→  1053→          <ArrowLeft className="mr-2 size-4" />
  1054→  1054→          Back
  1055→  1055→        </Button>
  1056→  1056→
  1057→  1057→        <div className="flex gap-2">
  1058→  1058→          {currentStep < 6 ? (
  1059→  1059→            <Button type="button" onClick={goNext} disabled={isSubmitting}>
  1060→  1060→              Next
  1061→  1061→              <ArrowRight className="ml-2 size-4" />
  1062→  1062→            </Button>
  1063→  1063→          ) : (
  1064→  1064→            <Button
  1065→  1065→              type="button"
  1066→  1066→              onClick={handleSubmit}
  1067→  1067→              disabled={isSubmitting}
  1068→  1068→              className="bg-teal-600 hover:bg-teal-700"
  1069→  1069→            >
  1070→  1070→              {isSubmitting ? (
  1071→  1071→                <>
  1072→  1072→                  <Loader2 className="mr-2 size-4 animate-spin" />
  1073→  1073→                  Submitting...
  1074→  1074→                </>
  1075→  1075→              ) : (
  1076→  1076→                <>
  1077→  1077→                  <Shield className="mr-2 size-4" />
  1078→  1078→                  Submit Application
  1079→  1079→                </>
  1080→  1080→              )}
  1081→  1081→            </Button>
  1082→  1082→          )}
  1083→  1083→        </div>
  1084→  1084→      </div>
  1085→  1085→
  1086→  1086→      {/* Loading overlay */}
  1087→  1087→      <AnimatePresence>
  1088→  1088→        {isSubmitting && (
  1089→  1089→          <motion.div
  1090→  1090→            initial={{ opacity: 0 }}
  1091→  1091→            animate={{ opacity: 1 }}
  1092→  1092→            exit={{ opacity: 0 }}
  1093→  1093→            className="fixed inset-0 z-50 flex flex-col items-center justify-center gap-4 bg-black/40 backdrop-blur-sm"
  1094→  1094→          >
  1095→  1095→            <motion.div
  1096→  1096→              animate={{ rotate: 360 }}
  1097→  1097→              transition={{ duration: 1, repeat: Infinity, ease: 'linear' }}
  1098→  1098→            >
  1099→  1099→              <Loader2 className="size-10 text-white" />
  1100→  1100→            </motion.div>
  1101→  1101→            <p className="text-sm font-medium text-white">Processing application...</p>
  1102→  1102→          </motion.div>
  1103→  1103→        )}
  1104→  1104→      </AnimatePresence>
  1105→  1105→
  1106→  1106→      {/* Success overlay */}
  1107→  1107→      {successResult && (
  1108→  1108→        <SuccessOverlay
  1109→  1109→          score={successResult.score}
  1110→  1110→          appNumber={successResult.appNumber}
  1111→  1111→          onClose={() => setSuccessResult(null)}
  1112→  1112→        />
  1113→  1113→      )}
  1114→  1114→    </div>
  1115→  1115→  );
  1116→  1116→}
