   150→   150→function maskAadhaar(aadhaar?: string) {
   151→   151→  if (!aadhaar || aadhaar.length < 4) return '••••••••••••';
   152→   152→  return `${aadhaar.slice(0, 4)}••••••••${aadhaar.slice(-4)}`;
   153→   153→}
   154→   154→
   155→   155→function formatDate(dateStr?: string | null) {
   156→   156→  if (!dateStr) return '—';
   157→   157→  return new Date(dateStr).toLocaleDateString('en-IN', {
   158→   158→    day: '2-digit', month: 'short', year: 'numeric',
   159→   159→  });
   160→   160→}
   161→   161→
   162→   162→function formatDateTime(dateStr?: string | null) {
   163→   163→  if (!dateStr) return '—';
   164→   164→  return new Date(dateStr).toLocaleString('en-IN', {
   165→   165→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   166→   166→  });
   167→   167→}
   168→   168→
   169→   169→function formatCurrency(amount: number) {
   170→   170→  return `\u20B9${amount.toLocaleString('en-IN')}`;
   171→   171→}
   172→   172→
   173→   173→function InfoRow({ label, value, icon: Icon }: { label: string; value: string | number | null | undefined; icon?: React.ElementType }) {
   174→   174→  return (
   175→   175→    <div className="flex items-center justify-between py-1.5 gap-2">

   178→   178→        {label}
   179→   179→      </span>
   180→   180→      <span className="text-sm font-medium text-right capitalize font-mono truncate">{value ?? '—'}</span>
   181→   181→    </div>
   182→   182→  );
   183→   183→}
