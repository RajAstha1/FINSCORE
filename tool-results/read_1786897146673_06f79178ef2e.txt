   683→          />
   684→          <Label htmlFor={item.name} className="text-sm leading-snug cursor-pointer">
   685→            {item.label}
   686→          </Label>
   687→        </div>
   688→      ))}
   689→      {(errors.consentCreditBureau || errors.consentDataSharing || errors.consentAccuracy) && (
   690→        <p className="text-xs text-destructive">
   691→          All consents are required to submit the application.
   692→        </p>
   693→      )}
   694→    </div>
   695→  );

   706→  appNumber: string;
   707→  onClose: () => void;

   712→    if (grade.includes('C')) return 'text-amber-600';
   713→    return 'text-red-600';

   741→            initial={{ scale: 0 }}

   746→            <CheckCircle2 className="size-10 text-emerald-600" />
   747→          </motion.div>

   788→// ─── Main Wizard ────────────────────────────────────────────────────────────
   789→
   790→export function ApplicationWizard() {
   791→  const [currentStep, setCurrentStep] = useState(1);
   792→  const [direction, setDirection] = useState<'forward' | 'backward'>('forward');
   793→  const [isSubmitting, setIsSubmitting] = useState(false);
   794→  const [submitError, setSubmitError] = useState<string | null>(null);
   795→  const [successResult, setSuccessResult] = useState<{
   796→    score: { totalScore: number; riskGrade: string; decisionType: string };
   797→    appNumber: string;
   798→  } | null>(null);
   799→
   800→  const token = useAuthStore((s) => s.token);
   801→  const navigate = useAppStore((s) => s.navigate);
   802→
   803→  const form = useForm<ApplicationFormData>({
   804→    resolver: zodResolver(fullSchema),
   805→    defaultValues: {
   806→      name: '',
   807→      aadhaarNumber: '',
   808→      panNumber: '',
   809→      dateOfBirth: '',
   810→      gender: '',
   811→      category: '',
   812→      educationLevel: '',
   813→      maritalStatus: '',
   814→      phone: '',
   815→      email: '',
   816→      address: '',
   817→      state: '',
   818→      district: '',
   819→      pincode: '',
   820→      occupation: '',
   821→      monthlyIncome: undefined,
   822→      bankName: '',
   823→      bankAccount: '',
   824→      bankIfsc: '',
   825→      loanAmount: undefined,
   826→      loanPurpose: '',
   827→      loanTenure: undefined,
   828→      schemeType: '',
   829→      electricityConsistency: 50,
   830→      mobileRechargeConsistency: 50,
   831→      utilityPaymentHistory: 50,
   832→      repaymentHistory: 50,
   833→      consentCreditBureau: false as unknown as true,
   834→      consentDataSharing: false as unknown as true,
   835→      consentAccuracy: false as unknown as true,
   836→    },
   837→    mode: 'onBlur',
   838→  });
   839→
   840→  const validateCurrentStep = useCallback(
   841→    async (step: number) => {
   842→      const fields = STEP_FIELDS[step];
   843→      if (!fields) return true;
   844→      return form.trigger(fields as string[]);
   845→    },
   846→    [form],
   847→  );
   848→
   849→  const goNext = useCallback(async () => {
   850→    const valid = await validateCurrentStep(currentStep);
   851→    if (valid) {
   852→      setDirection('forward');
   853→      setCurrentStep((s) => Math.min(s + 1, 6));
   854→    }
   855→  }, [currentStep, validateCurrentStep]);
   856→
   857→  const goBack = useCallback(() => {
   858→    setDirection('backward');
   859→    setCurrentStep((s) => Math.max(s - 1, 1));
   860→  }, []);
   861→
   862→  const handleSubmit = useCallback(async () => {
   863→    // Validate the last step (consent)
   864→    const valid = await validateCurrentStep(6);
   865→    if (!valid) return;
   866→
   867→    setIsSubmitting(true);
   868→    setSubmitError(null);
   869→
   870→    try {
   871→      const values = form.getValues();
   872→      const res = await fetch('/api/applications', {
   873→        method: 'POST',
   874→        headers: {
   875→          'Content-Type': 'application/json',
   876→          Authorization: `Bearer ${token}`,
   877→        },
   878→        body: JSON.stringify(values),
   879→      });
   880→
   881→      if (!res.ok) {
   882→        const errData = await res.json().catch(() => ({ error: 'Submission failed' }));
   883→        throw new Error(errData.error || `HTTP ${res.status}`);
   884→      }
   885→
   886→      const responseData = await res.json();
   887→      setSuccessResult({
   888→        score: responseData.score,
   889→        appNumber: responseData.application.applicationNumber,
   890→      });
   891→
   892→      // Navigate after 2 seconds
   893→      setTimeout(() => {
   894→        navigate('application-detail', { id: responseData.application.applicationNumber });
   895→      }, 2000);
   896→    } catch (err) {
   897→      setSubmitError(err instanceof Error ? err.message : 'Something went wrong');
   898→    } finally {
   899→      setIsSubmitting(false);
   900→    }
   901→  }, [form, token, validateCurrentStep, navigate]);
   902→
   903→  const progressValue = ((currentStep - 1) / 5) * 100;
   904→
   905→  const stepVariants = {
   906→    enter: (dir: 'forward' | 'backward') => ({
   907→      x: dir === 'forward' ? 300 : -300,
   908→      opacity: 0,
   909→    }),
   910→    center: {
   911→      x: 0,
   912→      opacity: 1,
   913→    },
   914→    exit: (dir: 'forward' | 'backward') => ({
   915→      x: dir === 'forward' ? -300 : 300,
   916→      opacity: 0,
   917→    }),
   918→  };
   919→
   920→  return (
   921→    <div className="mx-auto w-full max-w-4xl space-y-6">
   922→      {/* Header */}
   923→      <div>
   924→        <h1 className="text-2xl font-bold tracking-tight">New Loan Application</h1>
   925→        <p className="text-sm text-muted-foreground">Complete all steps to submit</p>
   926→      </div>
   927→
   928→      {/* Progress Bar */}
   929→      <Card className="overflow-hidden">
   930→        <CardContent className="p-4">
   931→          {/* Step indicators */}
   932→          <div className="mb-3 flex items-center justify-between">
   933→            {STEPS.map((step) => {
   934→              const StepIcon = step.icon;
   935→              const isCompleted = step.id < currentStep;
   936→              const isCurrent = step.id === currentStep;
   937→              return (
   938→                <div key={step.id} className="flex flex-1 flex-col items-center">
   939→                  <div
   940→                    className={`flex size-8 items-center justify-center rounded-full border-2 transition-colors ${
   941→                      isCompleted
   942→                        ? 'border-teal-500 bg-teal-500 text-white'
   943→                        : isCurrent
   944→                          ? 'border-teal-500 bg-white text-teal-600'
   945→                          : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   946→                    }`}
   947→                  >
   948→                    {isCompleted ? (
   949→                      <CheckCircle2 className="size-4" />
   950→                    ) : (
   951→                      <StepIcon className="size-4" />
   952→                    )}
   953→                  </div>
   954→                  <span
   955→                    className={`mt-1 hidden text-[10px] font-medium sm:block ${
   956→                      isCompleted || isCurrent ? 'text-teal-600' : 'text-muted-foreground'
   957→                    }`}
   958→                  >
   959→                    {step.name}
   960→                  </span>
   961→                </div>
   962→              );
   963→            })}
   964→          </div>
   965→          <Progress value={progressValue} className="h-2" />
   966→          <p className="mt-1.5 text-center text-xs text-muted-foreground">
   967→            Step {currentStep} of 6 — {STEPS[currentStep - 1].name}
   968→          </p>
   969→        </CardContent>
   970→      </Card>
   971→
   972→      {/* Step Card */}
   973→      <Form {...form}>
   974→      <Card>
   975→        <CardHeader className="pb-2">
   976→          <CardTitle className="flex items-center gap-2 text-base">
   977→            {(() => {
   978→              const StepIcon = STEPS[currentStep - 1].icon;
   979→              return <StepIcon className="size-5 text-teal-600" />;
   980→            })()}
   981→            {STEPS[currentStep - 1].name}
   982→          </CardTitle>
   983→        </CardHeader>
   984→        <CardContent className="pt-4">
   985→          <AnimatePresence mode="wait" custom={direction}>
   986→            <motion.div
   987→              key={currentStep}
   988→              custom={direction}
   989→              variants={stepVariants}
   990→              initial="enter"
   991→              animate="center"
   992→              exit="exit"
   993→              transition={{ duration: 0.25, ease: 'easeInOut' }}
   994→            >
   995→              {currentStep === 1 && <PersonalInfoStep />}
   996→              {currentStep === 2 && <AddressStep />}
   997→              {currentStep === 3 && <IncomeStep />}
   998→              {currentStep === 4 && <LoanDetailsStep />}
   999→              {currentStep === 5 && <ConsumptionStep />}
  1000→              {currentStep === 6 && <ReviewStep />}
  1001→            </motion.div>
  1002→          </AnimatePresence>
  1003→
  1004→          {submitError && (
  1005→            <motion.div
  1006→              initial={{ opacity: 0, y: 5 }}
  1007→              animate={{ opacity: 1, y: 0 }}
  1008→              className="mt-4 rounded-lg border border-destructive/30 bg-destructive/10 p-3 text-sm text-destructive"
  1009→            >
  1010→              {submitError}
  1011→            </motion.div>
  1012→          )}
  1013→        </CardContent>
  1014→      </Card>
  1015→      </Form>
  1016→
  1017→      {/* Navigation Buttons */}
  1018→      <div className="flex items-center justify-between">
  1019→        <Button
  1020→          type="button"
  1021→          variant="outline"
  1022→          onClick={goBack}
  1023→          disabled={currentStep === 1 || isSubmitting}
  1024→        >
  1025→          <ArrowLeft className="mr-2 size-4" />
  1026→          Back
  1027→        </Button>
  1028→
  1029→        <div className="flex gap-2">
  1030→          {currentStep < 6 ? (
  1031→            <Button type="button" onClick={goNext} disabled={isSubmitting}>
  1032→              Next
  1033→              <ArrowRight className="ml-2 size-4" />
  1034→            </Button>
  1035→          ) : (
  1036→            <Button
  1037→              type="button"
  1038→              onClick={handleSubmit}
  1039→              disabled={isSubmitting}
  1040→              className="bg-teal-600 hover:bg-teal-700"
  1041→            >
  1042→              {isSubmitting ? (
  1043→                <>
  1044→                  <Loader2 className="mr-2 size-4 animate-spin" />
  1045→                  Submitting...
  1046→                </>
  1047→              ) : (
  1048→                <>
  1049→                  <Shield className="mr-2 size-4" />
  1050→                  Submit Application
  1051→                </>
  1052→              )}
  1053→            </Button>
  1054→          )}
  1055→        </div>
  1056→      </div>
  1057→
  1058→      {/* Loading overlay */}
  1059→      <AnimatePresence>
  1060→        {isSubmitting && (
  1061→          <motion.div
  1062→            initial={{ opacity: 0 }}
  1063→            animate={{ opacity: 1 }}
  1064→            exit={{ opacity: 0 }}
  1065→            className="fixed inset-0 z-50 flex flex-col items-center justify-center gap-4 bg-black/40 backdrop-blur-sm"
  1066→          >
  1067→            <motion.div
  1068→              animate={{ rotate: 360 }}
  1069→              transition={{ duration: 1, repeat: Infinity, ease: 'linear' }}
  1070→            >
  1071→              <Loader2 className="size-10 text-white" />
  1072→            </motion.div>
  1073→            <p className="text-sm font-medium text-white">Processing application...</p>
  1074→          </motion.div>
  1075→        )}
  1076→      </AnimatePresence>
  1077→
  1078→      {/* Success overlay */}
  1079→      {successResult && (
  1080→        <SuccessOverlay
  1081→          score={successResult.score}
  1082→          appNumber={successResult.appNumber}
  1083→          onClose={() => setSuccessResult(null)}
  1084→        />
  1085→      )}
  1086→    </div>
  1087→  );
  1088→}
  1089→
  1090→export default ApplicationWizard;
