   833→   833→      consentCreditBureau: false as unknown as true,
   834→   834→      consentDataSharing: false as unknown as true,
   835→   835→      consentAccuracy: false as unknown as true,
   836→   836→    },
   837→   837→    mode: 'onBlur',
   838→   838→  });
   839→   839→
   840→   840→  const validateCurrentStep = useCallback(
   841→   841→    async (step: number) => {
   842→   842→      const fields = STEP_FIELDS[step];
   843→   843→      if (!fields) return true;
   844→   844→      return form.trigger(fields as string[]);
   845→   845→    },

   850→   850→    const valid = await validateCurrentStep(currentStep);
   851→   851→    if (valid) {
   852→   852→      setDirection('forward');
   853→   853→      setCurrentStep((s) => Math.min(s + 1, 6));
   854→   854→    }
   855→   855→  }, [currentStep, validateCurrentStep]);

   898→   898→    } finally {
   899→   899→      setIsSubmitting(false);
   900→   900→    }
   901→   901→  }, [form, token, validateCurrentStep, navigate]);
   902→   902→
   903→   903→  const progressValue = ((currentStep - 1) / 5) * 100;
   904→   904→
   905→   905→  const stepVariants = {
   906→   906→    enter: (dir: 'forward' | 'backward') => ({

   908→   908→      opacity: 0,
   909→   909→    }),
   910→   910→    center: {
   911→   911→      x: 0,
   912→   912→      opacity: 1,
   913→   913→    },
   914→   914→    exit: (dir: 'forward' | 'backward') => ({
   915→   915→      x: dir === 'forward' ? -300 : 300,
   916→   916→      opacity: 0,
   917→   917→    }),
   918→   918→  };
   919→   919→
   920→   920→  return (
   921→   921→    <div className="mx-auto w-full max-w-4xl space-y-6">
   922→   922→      {/* Header */}
   923→   923→      <div>
   924→   924→        <h1 className="text-2xl font-bold tracking-tight">New Loan Application</h1>
   925→   925→        <p className="text-sm text-muted-foreground">Complete all steps to submit</p>
   926→   926→      </div>
   927→   927→
   928→   928→      {/* Progress Bar */}
   929→   929→      <Card className="overflow-hidden">
   930→   930→        <CardContent className="p-4">
   931→   931→          {/* Step indicators */}
   932→   932→          <div className="mb-3 flex items-center justify-between">
   933→   933→            {STEPS.map((step) => {

   935→   935→              const isCompleted = step.id < currentStep;
   936→   936→              const isCurrent = step.id === currentStep;
   937→   937→              return (
   938→   938→                <div key={step.id} className="flex flex-1 flex-col items-center">
   939→   939→                  <div
   940→   940→                    className={`flex size-8 items-center justify-center rounded-full border-2 transition-colors ${
   941→   941→                      isCompleted
   942→   942→                        ? 'border-teal-500 bg-teal-500 text-white'
   943→   943→                        : isCurrent
   944→   944→                          ? 'border-teal-500 bg-white text-teal-600'
   945→   945→                          : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   946→   946→                    }`}
   947→   947→                  >
   948→   948→                    {isCompleted ? (
   949→   949→                      <CheckCircle2 className="size-4" />
   950→   950→                    ) : (
   951→   951→                      <StepIcon className="size-4" />
   952→   952→                    )}
   953→   953→                  </div>
   954→   954→                  <span
   955→   955→                    className={`mt-1 hidden text-[10px] font-medium sm:block ${
   956→   956→                      isCompleted || isCurrent ? 'text-teal-600' : 'text-muted-foreground'
   957→   957→                    }`}
   958→   958→                  >
   959→   959→                    {step.name}
   960→   960→                  </span>
   961→   961→                </div>
   962→   962→              );
   963→   963→            })}
   964→   964→          </div>
   965→   965→          <Progress value={progressValue} className="h-2" />
   966→   966→          <p className="mt-1.5 text-center text-xs text-muted-foreground">
   967→   967→            Step {currentStep} of 6 — {STEPS[currentStep - 1].name}
   968→   968→          </p>
   969→   969→        </CardContent>
   970→   970→      </Card>
   971→   971→
   972→   972→      {/* Step Card */}
   973→   973→      <Form {...form}>
   974→   974→      <Card>
   975→   975→        <CardHeader className="pb-2">
   976→   976→          <CardTitle className="flex items-center gap-2 text-base">
   977→   977→            {(() => {
   978→   978→              const StepIcon = STEPS[currentStep - 1].icon;
   979→   979→              return <StepIcon className="size-5 text-teal-600" />;
   980→   980→            })()}
   981→   981→            {STEPS[currentStep - 1].name}
   982→   982→          </CardTitle>
   983→   983→        </CardHeader>
   984→   984→        <CardContent className="pt-4">
   985→   985→          <AnimatePresence mode="wait" custom={direction}>
   986→   986→            <motion.div
   987→   987→              key={currentStep}
   988→   988→              custom={direction}
   989→   989→              variants={stepVariants}
   990→   990→              initial="enter"
   991→   991→              animate="center"
   992→   992→              exit="exit"
   993→   993→              transition={{ duration: 0.25, ease: 'easeInOut' }}
   994→   994→            >
   995→   995→              {currentStep === 1 && <PersonalInfoStep />}
   996→   996→              {currentStep === 2 && <AddressStep />}
   997→   997→              {currentStep === 3 && <IncomeStep />}
   998→   998→              {currentStep === 4 && <LoanDetailsStep />}
   999→   999→              {currentStep === 5 && <ConsumptionStep />}
  1000→  1000→              {currentStep === 6 && <ReviewStep />}
  1001→  1001→            </motion.div>
  1002→  1002→          </AnimatePresence>
  1003→  1003→
  1004→  1004→          {submitError && (
  1005→  1005→            <motion.div
  1006→  1006→              initial={{ opacity: 0, y: 5 }}
  1007→  1007→              animate={{ opacity: 1, y: 0 }}
  1008→  1008→              className="mt-4 rounded-lg border border-destructive/30 bg-destructive/10 p-3 text-sm text-destructive"
  1009→  1009→            >
  1010→  1010→              {submitError}
  1011→  1011→            </motion.div>
  1012→  1012→          )}
  1013→  1013→        </CardContent>
  1014→  1014→      </Card>
  1015→  1015→      </Form>
  1016→  1016→
  1017→  1017→      {/* Navigation Buttons */}
  1018→  1018→      <div className="flex items-center justify-between">
  1019→  1019→        <Button
  1020→  1020→          type="button"
  1021→  1021→          variant="outline"
  1022→  1022→          onClick={goBack}
  1023→  1023→          disabled={currentStep === 1 || isSubmitting}
  1024→  1024→        >
  1025→  1025→          <ArrowLeft className="mr-2 size-4" />
  1026→  1026→          Back
  1027→  1027→        </Button>
  1028→  1028→
  1029→  1029→        <div className="flex gap-2">
  1030→  1030→          {currentStep < 6 ? (
  1031→  1031→            <Button type="button" onClick={goNext} disabled={isSubmitting}>
  1032→  1032→              Next
  1033→  1033→              <ArrowRight className="ml-2 size-4" />
  1034→  1034→            </Button>
  1035→  1035→          ) : (
  1036→  1036→            <Button
  1037→  1037→              type="button"
  1038→  1038→              onClick={handleSubmit}
  1039→  1039→              disabled={isSubmitting}
  1040→  1040→              className="bg-teal-600 hover:bg-teal-700"
  1041→  1041→            >
  1042→  1042→              {isSubmitting ? (
  1043→  1043→                <>
  1044→  1044→                  <Loader2 className="mr-2 size-4 animate-spin" />
  1045→  1045→                  Submitting...
  1046→  1046→                </>
  1047→  1047→              ) : (
  1048→  1048→                <>
  1049→  1049→                  <Shield className="mr-2 size-4" />
  1050→  1050→                  Submit Application
  1051→  1051→                </>
  1052→  1052→              )}
  1053→  1053→            </Button>
  1054→  1054→          )}
  1055→  1055→        </div>
  1056→  1056→      </div>
  1057→  1057→
  1058→  1058→      {/* Loading overlay */}
  1059→  1059→      <AnimatePresence>
  1060→  1060→        {isSubmitting && (
  1061→  1061→          <motion.div
  1062→  1062→            initial={{ opacity: 0 }}
  1063→  1063→            animate={{ opacity: 1 }}
  1064→  1064→            exit={{ opacity: 0 }}
  1065→  1065→            className="fixed inset-0 z-50 flex flex-col items-center justify-center gap-4 bg-black/40 backdrop-blur-sm"
  1066→  1066→          >
  1067→  1067→            <motion.div
  1068→  1068→              animate={{ rotate: 360 }}
  1069→  1069→              transition={{ duration: 1, repeat: Infinity, ease: 'linear' }}
  1070→  1070→            >
  1071→  1071→              <Loader2 className="size-10 text-white" />
  1072→  1072→            </motion.div>
  1073→  1073→            <p className="text-sm font-medium text-white">Processing application...</p>
  1074→  1074→          </motion.div>
  1075→  1075→        )}
  1076→  1076→      </AnimatePresence>
  1077→  1077→
  1078→  1078→      {/* Success overlay */}
  1079→  1079→      {successResult && (
  1080→  1080→        <SuccessOverlay
  1081→  1081→          score={successResult.score}
  1082→  1082→          appNumber={successResult.appNumber}
  1083→  1083→          onClose={() => setSuccessResult(null)}
  1084→  1084→        />
  1085→  1085→      )}
  1086→  1086→    </div>
  1087→  1087→  );
  1088→  1088→}
