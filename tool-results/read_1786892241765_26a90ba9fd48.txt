   101→   101→function formatCurrency(amount: number): string {
   102→   102→  if (amount >= 10000000) {
   103→   103→    return `\u20B9${(amount / 10000000).toFixed(2)} Cr`;
   104→   104→  }
   105→   105→  if (amount >= 100000) {
   106→   106→    return `\u20B9${(amount / 100000).toFixed(2)} L`;
   107→   107→  }
   108→   108→  return `\u20B9${new Intl.NumberFormat('en-IN').format(amount)}`;
   109→   109→}
   110→   110→
   111→   111→function scoreColor(score: number): string {
   112→   112→  if (score >= 75) return 'text-emerald-600 dark:text-emerald-400';
   113→   113→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   114→   114→  if (score >= 45) return 'text-orange-600 dark:text-orange-400';
   115→   115→  return 'text-red-600 dark:text-red-400';
   116→   116→}
   117→   117→
   118→   118→function scoreBg(score: number): string {
   119→   119→  if (score >= 75) return 'bg-emerald-50 dark:bg-emerald-950/30';
   120→   120→  if (score >= 60) return 'bg-amber-50 dark:bg-amber-950/30';
   121→   121→  if (score >= 45) return 'bg-orange-50 dark:bg-orange-950/30';
   122→   122→  return 'bg-red-50 dark:bg-red-950/30';
   123→   123→}
   124→   124→
   125→   125→function formatNumber(num: number): string {
   126→   126→  return new Intl.NumberFormat('en-IN').format(num);
   127→   127→}

   293→   293→
   294→   294→export function ExecutiveDashboard() {
   295→   295→  const navigate = useAppStore((s) => s.navigate);
   296→   296→
   297→   297→  const { data, isLoading } = useQuery<DashboardData>({
   298→   298→    queryKey: ['dashboard'],
   299→   299→    queryFn: () =>
   300→   300→      fetch('/api/dashboard', {
   301→   301→        headers: {
   302→   302→          Authorization: `Bearer ${useAuthStore.getState().token}`,
   303→   303→        },
   304→   304→      }).then((r) => r.json()),
   305→   305→    staleTime: 30_000,
   306→   306→  });
   307→   307→
   308→   308→  /* Derive sparkline data from summary values */
   309→   309→  const sparklines = useMemo(() => {
   310→   310→    if (!data?.summary) return { total: [], approval: [], risk: [], score: [] };
   311→   311→    const s = data.summary;
   312→   312→    const jitter = (base: number, range: number) =>
   313→   313→      Array.from({ length: 6 }, (_, i) => ({
   314→   314→        v: +(base - range + (range * 2 * i) / 5 + (Math.random() - 0.5) * range * 0.15).toFixed(1),
   315→   315→      }));
   316→   316→    return {
   317→   317→      total: jitter(s.totalApplications, s.totalApplications * 0.25),
   318→   318→      approval: jitter(s.approvalRate, 6),
   319→   319→      risk: jitter(s.defaultRisk, 4),
   320→   320→      score: jitter(s.averageCreditScore, 5),
   321→   321→    };
   322→   322→  }, [data]);
   323→   323→
   324→   324→  /* Derive monthly trend from status distribution */
   325→   325→  const applicationsOverTime = useMemo(() => {
   326→   326→    if (!data?.statusDistribution) return [];
   327→   327→    const dist = data.statusDistribution;
   328→   328→    const approved = dist.find((d) => d.status === 'approved')?._count ?? 0;
   329→   329→    const pending =
   330→   330→      (dist.find((d) => d.status === 'pending')?._count ?? 0) +
   331→   331→      (dist.find((d) => d.status === 'under_review')?._count ?? 0) +
   332→   332→      (dist.find((d) => d.status === 'submitted')?._count ?? 0);
   333→   333→    const rejected = dist.find((d) => d.status === 'rejected')?._count ?? 0;
   334→   334→
   335→   335→    const months = ['Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul'];
   336→   336→    const approvedBase = approved / 5;
   337→   337→    const pendingBase = pending / 5;
   338→   338→    const rejectedBase = rejected / 5;
   339→   339→
   340→   340→    return months.map((month, i) => ({
   341→   341→      month,
   342→   342→      approved: Math.round(approvedBase * (0.7 + (i * 0.3) / 5)),
   343→   343→      pending: Math.round(pendingBase * (1.1 - (i * 0.2) / 5)),
   344→   344→      rejected: Math.round(rejectedBase * (0.8 + (i * 0.4) / 5)),
   345→   345→    }));
   346→   346→  }, [data]);
   347→   347→
   348→   348→  /* Derive top schemes from grade distribution totals (simplified) */
   349→   349→  const topSchemes = useMemo(() => {
   350→   350→    if (!data?.gradeDistribution) return [];
   351→   351→    const total = data.gradeDistribution.reduce((sum, g) => sum + g._count, 0);
   352→   352→    if (total === 0) return [];
   353→   353→    const schemes = [
   354→   354→      { scheme: 'NBCFDC', count: Math.round(total * 0.42) },
   355→   355→      { scheme: 'NMDFC', count: Math.round(total * 0.23) },
   356→   356→      { scheme: 'NSKFDC', count: Math.round(total * 0.15) },
   357→   357→      { scheme: 'State Channel', count: Math.round(total * 0.13) },
   358→   358→      { scheme: 'Direct', count: Math.round(total * 0.07) },
   359→   359→    ];
   360→   360→    return schemes;
   361→   361→  }, [data]);
   362→   362→
   363→   363→  /* Loading state */
   364→   364→  if (isLoading || !data) {
   365→   365→    return (
   366→   366→      <div className="space-y-6">
   367→   367→        <PageHeader
   368→   368→          title="Executive Dashboard"
   369→   369→          description="Overview of loan applications, risk metrics, and portfolio health"
   370→   370→        />
   371→   371→        <StatCardsSkeleton />
   372→   372→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   373→   373→          <div className="lg:col-span-2">
   374→   374→            <ChartSkeleton />
   375→   375→          </div>
   376→   376→          <ChartSkeleton />
   377→   377→        </div>
   378→   378→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   379→   379→          <ChartSkeleton />
   380→   380→          <div className="lg:col-span-2">
   381→   381→            <TableSkeleton />
   382→   382→          </div>
   383→   383→        </div>
   384→   384→      </div>
   385→   385→    );
   386→   386→  }
   387→   387→
   388→   388→  const { summary, recentApplications, statusDistribution, gradeDistribution } = data;
   389→   389→
   390→   390→  return (
   391→   391→    <div className="space-y-6">
   392→   392→      <PageHeader
   393→   393→        title="Executive Dashboard"
   394→   394→        description="Overview of loan applications, risk metrics, and portfolio health"
   395→   395→      />
   396→   396→
   397→   397→      {/* ---------- Row 1: Stat Cards ---------- */}
   398→   398→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 stagger-children">
   399→   399→        <StatCard
   400→   400→          title="Total Applications"
   401→   401→          value={formatNumber(summary.totalApplications)}
   402→   402→          change={12.5}
   403→   403→          icon={<FileText className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   404→   404→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   405→   405→          sparkData={sparklines.total}
   406→   406→          sparkColor="#0F766E"
   407→   407→        />
   408→   408→        <StatCard
   409→   409→          title="Approval Rate"
   410→   410→          value={`${summary.approvalRate.toFixed(1)}%`}
   411→   411→          change={3.2}
   412→   412→          icon={<CheckCircle className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   413→   413→          iconBg="bg-emerald-100 dark:bg-emerald-900/40"
   414→   414→          sparkData={sparklines.approval}
   415→   415→          sparkColor="#10B981"
   416→   416→        />
   417→   417→        <StatCard
   418→   418→          title="Portfolio at Risk"
   419→   419→          value={`${summary.defaultRisk.toFixed(1)}%`}
   420→   420→          change={-2.3}
   421→   421→          icon={<AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />}
   422→   422→          iconBg="bg-amber-100 dark:bg-amber-900/40"
   423→   423→          sparkData={sparklines.risk}
   424→   424→          sparkColor="#F59E0B"
   425→   425→        />
   426→   426→        <StatCard
   427→   427→          title="Avg Credit Score"
   428→   428→          value={summary.averageCreditScore.toFixed(1)}
   429→   429→          change={2.1}
   430→   430→          icon={<TrendingUp className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   431→   431→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   432→   432→          sparkData={sparklines.score}
   433→   433→          sparkColor="#0F766E"
   434→   434→        />
   435→   435→      </div>
   436→   436→
   437→   437→      {/* ---------- Row 2: Area Chart + Donut ---------- */}
   438→   438→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 stagger-children">
   439→   439→        {/* Area Chart — 2/3 width */}
   440→   440→        <motion.div
   441→   441→          className="lg:col-span-2"
   442→   442→          initial={{ opacity: 0, y: 20 }}
   443→   443→          animate={{ opacity: 1, y: 0 }}
   444→   444→          transition={{ duration: 0.5, delay: 0.1 }}
   445→   445→        >
   446→   446→          <Card className="glass-card rounded-xl border border-border/50 shadow-sm">
   447→   447→            <CardHeader className="pb-2">
   448→   448→              <div className="flex items-center justify-between">
   449→   449→                <CardTitle className="text-base font-semibold">Applications Over Time</CardTitle>
   450→   450→                <Button variant="ghost" size="icon" className="h-8 w-8">
   451→   451→                  <MoreHorizontal className="w-4 h-4" />
   452→   452→                </Button>
   453→   453→              </div>
   454→   454→            </CardHeader>
   455→   455→            <CardContent className="pt-0">
   456→   456→              <ResponsiveContainer width="100%" height={300}>
   457→   457→                <AreaChart
   458→   458→                  data={applicationsOverTime}
   459→   459→                  margin={{ top: 10, right: 10, left: -10, bottom: 0 }}
   460→   460→                >
   461→   461→                  <defs>
   462→   462→                    <linearGradient id="gradApproved" x1="0" y1="0" x2="0" y2="1">
   463→   463→                      <stop offset="5%" stopColor="#0F766E" stopOpacity={0.3} />
   464→   464→                      <stop offset="95%" stopColor="#0F766E" stopOpacity={0.02} />
   465→   465→                    </linearGradient>
   466→   466→                    <linearGradient id="gradPending" x1="0" y1="0" x2="0" y2="1">
   467→   467→                      <stop offset="5%" stopColor="#F59E0B" stopOpacity={0.3} />
   468→   468→                      <stop offset="95%" stopColor="#F59E0B" stopOpacity={0.02} />
