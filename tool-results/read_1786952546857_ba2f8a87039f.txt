   818→export function ApplicationWizard() {
   819→  const [currentStep, setCurrentStep] = useState(1);
   820→  const [direction, setDirection] = useState<'forward' | 'backward'>('forward');
   821→  const [isSubmitting, setIsSubmitting] = useState(false);
   822→  const [submitError, setSubmitError] = useState<string | null>(null);
   823→  const [successResult, setSuccessResult] = useState<{
   824→    score: { totalScore: number; riskGrade: string; decisionType: string };
   825→    appNumber: string;
   826→  } | null>(null);
   827→
   828→  const token = useAuthStore((s) => s.token);
   829→  const navigate = useAppStore((s) => s.navigate);
   830→
   831→  const form = useForm<ApplicationFormData>({
   832→    resolver: zodResolver(fullSchema),
   833→    defaultValues: {
   834→      name: '',
   835→      aadhaarNumber: '',
   836→      panNumber: '',
   837→      dateOfBirth: '',
   838→      gender: '',
   839→      category: '',
   840→      educationLevel: '',
   841→      maritalStatus: '',
   842→      phone: '',
   843→      email: '',
   844→      address: '',
   845→      state: '',
   846→      district: '',
   847→      pincode: '',
   848→      occupation: '',
   849→      monthlyIncome: undefined,
   850→      bankName: '',
   851→      bankAccount: '',
   852→      bankIfsc: '',
   853→      loanAmount: undefined,
   854→      loanPurpose: '',
   855→      loanTenure: undefined,
   856→      schemeType: '',
   857→      electricityConsistency: 50,
   858→      mobileRechargeConsistency: 50,
   859→      utilityPaymentHistory: 50,
   860→      repaymentHistory: 50,
   861→      consentCreditBureau: false,
   862→      consentDataSharing: false,
   863→      consentAccuracy: false,
   864→    },
   865→    mode: 'onBlur',
   866→  });
   867→
   868→  const validateCurrentStep = useCallback(
   869→    async (step: number) => {
   870→      const fields = STEP_FIELDS[step];
   871→      if (!fields) return true;
   872→      return form.trigger(fields as string[]);
   873→    },
   874→    [form],
   875→  );
   876→
   877→  const goNext = useCallback(async () => {
   878→    const valid = await validateCurrentStep(currentStep);
   879→    if (valid) {
   880→      setDirection('forward');
   881→      setCurrentStep((s) => Math.min(s + 1, 6));
   882→    }
   883→  }, [currentStep, validateCurrentStep]);
   884→
   885→  const goBack = useCallback(() => {
   886→    setDirection('backward');
   887→    setCurrentStep((s) => Math.max(s - 1, 1));
   888→  }, []);
   889→
   890→  const handleSubmit = useCallback(async () => {
   891→    // Validate the last step (consent)
   892→    const valid = await validateCurrentStep(6);
   893→    if (!valid) return;
   894→
   895→    setIsSubmitting(true);
   896→    setSubmitError(null);
   897→
   898→    try {
   899→      const values = form.getValues();
   900→      const res = await fetch('/api/applications', {
   901→        method: 'POST',
   902→        headers: {
   903→          'Content-Type': 'application/json',
   904→          Authorization: `Bearer ${token}`,
   905→        },
   906→        body: JSON.stringify(values),
   907→      });
   908→
   909→      if (!res.ok) {
   910→        const errData = await res.json().catch(() => ({ error: 'Submission failed' }));
   911→        throw new Error(errData.error || `HTTP ${res.status}`);
   912→      }
   913→
   914→      const responseData = await res.json();
   915→      setSuccessResult({
   916→        score: responseData.score,
   917→        appNumber: responseData.application.applicationNumber,
   918→      });
   919→
   920→      // Navigate after 2 seconds
   921→      setTimeout(() => {
   922→        navigate('application-detail', { id: responseData.application.id });
   923→      }, 2000);
   924→    } catch (err) {
   925→      setSubmitError(err instanceof Error ? err.message : 'Something went wrong');
   926→    } finally {
   927→      setIsSubmitting(false);
   928→    }
   929→  }, [form, token, validateCurrentStep, navigate]);
   930→
   931→  const progressValue = ((currentStep - 1) / 5) * 100;
   932→
   933→  const stepVariants = {
   934→    enter: (dir: 'forward' | 'backward') => ({
   935→      x: dir === 'forward' ? 300 : -300,
   936→      opacity: 0,
   937→    }),
   938→    center: {
   939→      x: 0,
   940→      opacity: 1,
   941→    },
   942→    exit: (dir: 'forward' | 'backward') => ({
   943→      x: dir === 'forward' ? -300 : 300,
   944→      opacity: 0,
   945→    }),
   946→  };
   947→
   948→  return (
   949→    <div className="mx-auto w-full max-w-4xl space-y-6">
   950→      {/* Header */}
   951→      <div>
   952→        <h1 className="text-2xl font-bold tracking-tight">New Loan Application</h1>
   953→        <p className="text-sm text-muted-foreground">Complete all steps to submit</p>
   954→      </div>
   955→
   956→      {/* Progress Bar */}
   957→      <Card className="overflow-hidden">
   958→        <CardContent className="p-4">
   959→          {/* Step indicators */}
   960→          <div className="mb-3 flex items-center justify-between">
   961→            {STEPS.map((step) => {
   962→              const StepIcon = step.icon;
   963→              const isCompleted = step.id < currentStep;
   964→              const isCurrent = step.id === currentStep;

   966→                <div key={step.id} className="flex flex-1 flex-col items-center">
   967→                  <div
   968→                    className={`flex size-8 items-center justify-center rounded-full border-2 transition-colors ${
   969→                      isCompleted
   970→                        ? 'border-teal-500 bg-teal-500 text-white'
   971→                        : isCurrent
   972→                          ? 'border-teal-500 bg-white text-teal-600'
   973→                          : 'border-muted-foreground/30 bg-muted text-muted-foreground'
   974→                    }`}
   975→                  >
   976→                    {isCompleted ? (
   977→                      <CheckCircle2 className="size-4" />
   978→                    ) : (
   979→                      <StepIcon className="size-4" />
   980→                    )}
   981→                  </div>
   982→                  <span
   983→                    className={`mt-1 hidden text-[10px] font-medium sm:block ${

   986→                  >
   987→                    {step.name}
   988→                  </span>
   989→                </div>
   990→              );
   991→            })}
   992→          </div>
   993→          <Progress value={progressValue} className="h-2" />
   994→          <p className="mt-1.5 text-center text-xs text-muted-foreground">
   995→            Step {currentStep} of 6 — {STEPS[currentStep - 1].name}
   996→          </p>
   997→        </CardContent>
   998→      </Card>
   999→
  1000→      {/* Step Card */}
  1001→      <Form {...form}>
  1002→      <Card>
  1003→        <CardHeader className="pb-2">
  1004→          <CardTitle className="flex items-center gap-2 text-base">
  1005→            {(() => {
  1006→              const StepIcon = STEPS[currentStep - 1].icon;
  1007→              return <StepIcon className="size-5 text-teal-600" />;
  1008→            })()}
  1009→            {STEPS[currentStep - 1].name}
  1010→          </CardTitle>
  1011→        </CardHeader>
  1012→        <CardContent className="pt-4">
  1013→          <AnimatePresence mode="wait" custom={direction}>
  1014→            <motion.div
  1015→              key={currentStep}
  1016→              custom={direction}
  1017→              variants={stepVariants}
  1018→              initial="enter"
  1019→              animate="center"
  1020→              exit="exit"
  1021→              transition={{ duration: 0.25, ease: 'easeInOut' }}
  1022→            >
  1023→              {currentStep === 1 && <PersonalInfoStep />}
  1024→              {currentStep === 2 && <AddressStep />}
  1025→              {currentStep === 3 && <IncomeStep />}
  1026→              {currentStep === 4 && <LoanDetailsStep />}

  1028→              {currentStep === 6 && <ReviewStep />}
