    99→function formatCurrency(amount: number): string {
   100→  if (amount >= 10000000) {
   101→    return `\u20B9${(amount / 10000000).toFixed(2)} Cr`;
   102→  }
   103→  if (amount >= 100000) {
   104→    return `\u20B9${(amount / 100000).toFixed(2)} L`;
   105→  }
   106→  return `\u20B9${new Intl.NumberFormat('en-IN').format(amount)}`;
   107→}
   108→
   109→function scoreColor(score: number): string {
   110→  if (score >= 75) return 'text-emerald-600 dark:text-emerald-400';
   111→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   112→  if (score >= 45) return 'text-orange-600 dark:text-orange-400';
   113→  return 'text-red-600 dark:text-red-400';
   114→}
   115→
   116→function scoreBg(score: number): string {
   117→  if (score >= 75) return 'bg-emerald-50 dark:bg-emerald-950/30';
   118→  if (score >= 60) return 'bg-amber-50 dark:bg-amber-950/30';
   119→  if (score >= 45) return 'bg-orange-50 dark:bg-orange-950/30';
   120→  return 'bg-red-50 dark:bg-red-950/30';
   121→}
   122→
   123→function scoreBorderColor(score: number): string {
   124→  if (score >= 75) return 'border-l-emerald-500 dark:border-l-emerald-400';
   125→  if (score >= 60) return 'border-l-amber-500 dark:border-l-amber-400';
   126→  if (score >= 45) return 'border-l-orange-500 dark:border-l-orange-400';
   127→  return 'border-l-red-500 dark:border-l-red-400';
   128→}

   253→
   254→interface PieLabelProps {
   255→  cx: number;
   256→  cy: number;
   257→  midAngle: number;
   258→  innerRadius: number;
   259→  outerRadius: number;
   260→  percent: number;
   261→  riskGrade: string;
   262→}
   263→
   264→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent }: PieLabelProps) {
   265→  if (percent < 0.05) return null;
   266→  const RADIAN = Math.PI / 180;
   267→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   268→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   269→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   270→  return (
   271→    <text
   272→      x={x}
   273→      y={y}
   274→      fill="white"
   275→      textAnchor="middle"
   276→      dominantBaseline="central"
   277→      fontSize={11}
   278→      fontWeight={600}
   279→    >
   280→      {`${(percent * 100).toFixed(0)}%`}
   281→    </text>
   282→  );
   283→}

   346→export function ExecutiveDashboard() {
   347→  const navigate = useAppStore((s) => s.navigate);
   348→
   349→  const { data, isLoading } = useQuery<DashboardData>({
   350→    queryKey: ['dashboard'],
   351→    queryFn: () =>
   352→      fetch('/api/dashboard', {
   353→        headers: {
   354→          Authorization: `Bearer ${useAuthStore.getState().token}`,
   355→        },
   356→      }).then((r) => r.json()),
   357→    staleTime: 30_000,
   358→  });
   359→
   360→  /* Derive sparkline data from summary values */
   361→  const sparklines = useMemo(() => {
   362→    if (!data?.summary) return { total: [], approval: [], risk: [], score: [] };
   363→    const s = data.summary;
   364→    const jitter = (base: number, range: number) =>
   365→      Array.from({ length: 6 }, (_, i) => ({
   366→        v: +(base - range + (range * 2 * i) / 5 + (Math.random() - 0.5) * range * 0.15).toFixed(1),
   367→      }));
   368→    return {
   369→      total: jitter(s.totalApplications, s.totalApplications * 0.25),
   370→      approval: jitter(s.approvalRate, 6),
   371→      risk: jitter(s.defaultRisk, 4),
   372→      score: jitter(s.averageCreditScore, 5),
   373→    };
   374→  }, [data]);
   375→
   376→  /* Derive monthly trend from status distribution */
   377→  const applicationsOverTime = useMemo(() => {
   378→    if (!data?.statusDistribution) return [];
   379→    const dist = data.statusDistribution;
   380→    const approved = dist.find((d) => d.status === 'approved')?._count ?? 0;
   381→    const pending =
   382→      (dist.find((d) => d.status === 'pending')?._count ?? 0) +
   383→      (dist.find((d) => d.status === 'under_review')?._count ?? 0) +
   384→      (dist.find((d) => d.status === 'submitted')?._count ?? 0);
   385→    const rejected = dist.find((d) => d.status === 'rejected')?._count ?? 0;
   386→
   387→    const months = ['Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul'];
   388→    const approvedBase = approved / 5;
   389→    const pendingBase = pending / 5;
   390→    const rejectedBase = rejected / 5;
   391→
   392→    return months.map((month, i) => ({
   393→      month,
   394→      approved: Math.round(approvedBase * (0.7 + (i * 0.3) / 5)),
   395→      pending: Math.round(pendingBase * (1.1 - (i * 0.2) / 5)),
   396→      rejected: Math.round(rejectedBase * (0.8 + (i * 0.4) / 5)),
   397→    }));
   398→  }, [data]);
   399→
   400→  /* Derive top schemes from grade distribution totals (simplified) */
   401→  const topSchemes = useMemo(() => {
   402→    if (!data?.gradeDistribution) return [];
   403→    const total = data.gradeDistribution.reduce((sum, g) => sum + g._count, 0);
   404→    if (total === 0) return [];
   405→    const schemes = [
   406→      { scheme: 'NBCFDC', count: Math.round(total * 0.42) },
   407→      { scheme: 'NMDFC', count: Math.round(total * 0.23) },
   408→      { scheme: 'NSKFDC', count: Math.round(total * 0.15) },
   409→      { scheme: 'State Channel', count: Math.round(total * 0.13) },
   410→      { scheme: 'Direct', count: Math.round(total * 0.07) },
   411→    ];
   412→    return schemes;
   413→  }, [data]);
   414→
   415→  /* Total count for donut center label */
   416→  const gradeTotal = useMemo(() => {
   417→    if (!data?.gradeDistribution) return 0;
   418→    return data.gradeDistribution.reduce((sum, g) => sum + g._count, 0);
   419→  }, [data]);
   420→
   421→  /* Loading state */
   422→  if (isLoading || !data) {
   423→    return (
   424→      <div className="space-y-6">
   425→        <PageHeader
   426→          title="Executive Dashboard"
   427→          description="Overview of loan applications, risk metrics, and portfolio health"
   428→        />
   429→        <StatCardsSkeleton />
   430→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   431→          <div className="lg:col-span-2">
   432→            <ChartSkeleton />
   433→          </div>
   434→          <ChartSkeleton />
   435→        </div>
   436→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   437→          <ChartSkeleton />
   438→          <div className="lg:col-span-2">
   439→            <TableSkeleton />
   440→          </div>
   441→        </div>
   442→      </div>
   443→    );
   444→  }
   445→
   446→  const { summary, recentApplications, statusDistribution, gradeDistribution } = data;
   447→
   448→  return (
   449→    <div className="space-y-6">
   450→      <PageHeader
   451→        title="Executive Dashboard"
   452→        description="Overview of loan applications, risk metrics, and portfolio health"
   453→      />
   454→
   455→      {/* ---------- Row 1: Stat Cards ---------- */}
   456→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 stagger-children">
   457→        <StatCard
   458→          title="Total Applications"
   459→          value={formatNumber(summary.totalApplications)}
   460→          change={12.5}
   461→          icon={<FileText className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   462→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   463→          sparkData={sparklines.total}
   464→          sparkColor="#0F766E"
   465→        />
   466→        <StatCard
   467→          title="Approval Rate"
   468→          value={`${summary.approvalRate.toFixed(1)}%`}
   469→          change={3.2}
   470→          icon={<CheckCircle className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   471→          iconBg="bg-emerald-100 dark:bg-emerald-900/40"
   472→          sparkData={sparklines.approval}
   473→          sparkColor="#10B981"
   474→        />
   475→        <StatCard
   476→          title="Portfolio at Risk"
   477→          value={`${summary.defaultRisk.toFixed(1)}%`}
   478→          change={-2.3}
   479→          icon={<AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />}
   480→          iconBg="bg-amber-100 dark:bg-amber-900/40"
   481→          sparkData={sparklines.risk}
   482→          sparkColor="#F59E0B"
   483→        />
   484→        <StatCard
   485→          title="Avg Credit Score"
   486→          value={summary.averageCreditScore.toFixed(1)}
   487→          change={2.1}
   488→          icon={<TrendingUp className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   489→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   490→          sparkData={sparklines.score}
   491→          sparkColor="#0F766E"
   492→        />
   493→      </div>
   494→
   495→      {/* ---------- Row 2: Area Chart + Donut ---------- */}
   496→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 stagger-children">
   497→        {/* Area Chart \u2014 2/3 width */}

   504→          {/* [9] Added hover:shadow-md, removed MoreHorizontal button */}

   506→            <CardHeader className="pb-2">

   509→              <div className="flex items-center gap-4 text-xs pt-1">
   510→                <span className="flex items-center gap-1.5">
   511→                  <span className="w-2.5 h-2.5 rounded-full bg-[#0F766E]" />
   512→                  Approved
   513→                </span>
   514→                <span className="flex items-center gap-1.5">
   515→                  <span className="w-2.5 h-2.5 rounded-full bg-[#F59E0B]" />
   516→                  Pending
   517→                </span>
