   151→function maskAadhaar(aadhaar?: string) {
   152→  if (!aadhaar || aadhaar.length < 4) return '••••••••••••';
   153→  return `${aadhaar.slice(0, 4)}••••••••${aadhaar.slice(-4)}`;
   154→}
   155→
   156→function formatDate(dateStr?: string | null) {
   157→  if (!dateStr) return '—';
   158→  return new Date(dateStr).toLocaleDateString('en-IN', {
   159→    day: '2-digit', month: 'short', year: 'numeric',
   160→  });
   161→}
   162→
   163→function formatDateTime(dateStr?: string | null) {
   164→  if (!dateStr) return '—';
   165→  return new Date(dateStr).toLocaleString('en-IN', {
   166→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   167→  });
   168→}
   169→
   170→function formatCurrency(amount: number) {
   171→  return `\u20B9${amount.toLocaleString('en-IN')}`;
   172→}
   173→
   174→function InfoRow({ label, value, icon: Icon }: { label: string; value: string | number | null | undefined; icon?: React.ElementType }) {
   175→  return (

   181→      <span className="text-sm font-medium text-right capitalize font-mono truncate">{value ?? '—'}</span>
   182→    </div>
   183→  );
   184→}
