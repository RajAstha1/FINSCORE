   101→function formatCurrency(amount: number): string {
   102→  if (amount >= 10000000) {
   103→    return `\u20B9${(amount / 10000000).toFixed(2)} Cr`;
   104→  }
   105→  if (amount >= 100000) {
   106→    return `\u20B9${(amount / 100000).toFixed(2)} L`;
   107→  }
   108→  return `\u20B9${new Intl.NumberFormat('en-IN').format(amount)}`;
   109→}
   110→
   111→function scoreColor(score: number): string {
   112→  if (score >= 75) return 'text-emerald-600 dark:text-emerald-400';
   113→  if (score >= 60) return 'text-amber-600 dark:text-amber-400';
   114→  if (score >= 45) return 'text-orange-600 dark:text-orange-400';
   115→  return 'text-red-600 dark:text-red-400';
   116→}
   117→
   118→function scoreBg(score: number): string {
   119→  if (score >= 75) return 'bg-emerald-50 dark:bg-emerald-950/30';
   120→  if (score >= 60) return 'bg-amber-50 dark:bg-amber-950/30';
   121→  if (score >= 45) return 'bg-orange-50 dark:bg-orange-950/30';
   122→  return 'bg-red-50 dark:bg-red-950/30';
   123→}
   124→
   125→function formatNumber(num: number): string {
   126→  return new Intl.NumberFormat('en-IN').format(num);
   127→}

   209→function CustomAreaTooltip({
   210→  active,
   211→  payload,
   212→  label,
   213→}: {
   214→  active?: boolean;
   215→  payload?: Array<{ name: string; value: number; color: string }>;
   216→  label?: string;
   217→}) {

   220→    <div className="rounded-lg border border-border bg-card p-3 shadow-lg">
   221→      <p className="text-sm font-medium mb-1.5">{label}</p>

   223→        <div key={item.name} className="flex items-center gap-2 text-xs">
   224→          <span className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: item.color }} />
   225→          <span className="text-muted-foreground capitalize">{item.name.replace('_', ' ')}:</span>
   226→          <span className="font-mono font-medium">{item.value}</span>
   227→        </div>
   228→      ))}

   294→export function ExecutiveDashboard() {
   295→  const navigate = useAppStore((s) => s.navigate);
   296→
   297→  const { data, isLoading } = useQuery<DashboardData>({
   298→    queryKey: ['dashboard'],
   299→    queryFn: () =>
   300→      fetch('/api/dashboard', {
   301→        headers: {
   302→          Authorization: `Bearer ${useAuthStore.getState().token}`,
   303→        },
   304→      }).then((r) => r.json()),
   305→    staleTime: 30_000,
   306→  });
   307→
   308→  /* Derive sparkline data from summary values */
   309→  const sparklines = useMemo(() => {
   310→    if (!data?.summary) return { total: [], approval: [], risk: [], score: [] };
   311→    const s = data.summary;
   312→    const jitter = (base: number, range: number) =>
   313→      Array.from({ length: 6 }, (_, i) => ({
   314→        v: +(base - range + (range * 2 * i) / 5 + (Math.random() - 0.5) * range * 0.15).toFixed(1),
   315→      }));
   316→    return {
   317→      total: jitter(s.totalApplications, s.totalApplications * 0.25),
   318→      approval: jitter(s.approvalRate, 6),
   319→      risk: jitter(s.defaultRisk, 4),
   320→      score: jitter(s.averageCreditScore, 5),
   321→    };
   322→  }, [data]);
   323→
   324→  /* Derive monthly trend from status distribution */
   325→  const applicationsOverTime = useMemo(() => {
   326→    if (!data?.statusDistribution) return [];
   327→    const dist = data.statusDistribution;
   328→    const approved = dist.find((d) => d.status === 'approved')?._count ?? 0;
   329→    const pending =
   330→      (dist.find((d) => d.status === 'pending')?._count ?? 0) +
   331→      (dist.find((d) => d.status === 'under_review')?._count ?? 0) +
   332→      (dist.find((d) => d.status === 'submitted')?._count ?? 0);
   333→    const rejected = dist.find((d) => d.status === 'rejected')?._count ?? 0;
   334→
   335→    const months = ['Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul'];
   336→    const approvedBase = approved / 5;
   337→    const pendingBase = pending / 5;
   338→    const rejectedBase = rejected / 5;
   339→
   340→    return months.map((month, i) => ({
   341→      month,
   342→      approved: Math.round(approvedBase * (0.7 + (i * 0.3) / 5)),
   343→      pending: Math.round(pendingBase * (1.1 - (i * 0.2) / 5)),
   344→      rejected: Math.round(rejectedBase * (0.8 + (i * 0.4) / 5)),
   345→    }));
   346→  }, [data]);
   347→
   348→  /* Derive top schemes from grade distribution totals (simplified) */
   349→  const topSchemes = useMemo(() => {
   350→    if (!data?.gradeDistribution) return [];
   351→    const total = data.gradeDistribution.reduce((sum, g) => sum + g._count, 0);
   352→    if (total === 0) return [];
   353→    const schemes = [
   354→      { scheme: 'NBCFDC', count: Math.round(total * 0.42) },
   355→      { scheme: 'NMDFC', count: Math.round(total * 0.23) },
   356→      { scheme: 'NSKFDC', count: Math.round(total * 0.15) },
   357→      { scheme: 'State Channel', count: Math.round(total * 0.13) },
   358→      { scheme: 'Direct', count: Math.round(total * 0.07) },
   359→    ];
   360→    return schemes;
   361→  }, [data]);
   362→
   363→  /* Loading state */
   364→  if (isLoading || !data) {
   365→    return (
   366→      <div className="space-y-6">
   367→        <PageHeader
   368→          title="Executive Dashboard"
   369→          description="Overview of loan applications, risk metrics, and portfolio health"
   370→        />
   371→        <StatCardsSkeleton />
   372→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   373→          <div className="lg:col-span-2">
   374→            <ChartSkeleton />
   375→          </div>
   376→          <ChartSkeleton />
   377→        </div>
   378→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   379→          <ChartSkeleton />
   380→          <div className="lg:col-span-2">
   381→            <TableSkeleton />
   382→          </div>
   383→        </div>
   384→      </div>
   385→    );
   386→  }
   387→
   388→  const { summary, recentApplications, statusDistribution, gradeDistribution } = data;
   389→
   390→  return (
   391→    <div className="space-y-6">
   392→      <PageHeader
   393→        title="Executive Dashboard"
   394→        description="Overview of loan applications, risk metrics, and portfolio health"
   395→      />
   396→
   397→      {/* ---------- Row 1: Stat Cards ---------- */}
   398→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 stagger-children">
   399→        <StatCard
   400→          title="Total Applications"
   401→          value={formatNumber(summary.totalApplications)}
   402→          change={12.5}
   403→          icon={<FileText className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   404→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   405→          sparkData={sparklines.total}
   406→          sparkColor="#0F766E"
   407→        />
   408→        <StatCard
   409→          title="Approval Rate"
   410→          value={`${summary.approvalRate.toFixed(1)}%`}
   411→          change={3.2}
   412→          icon={<CheckCircle className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   413→          iconBg="bg-emerald-100 dark:bg-emerald-900/40"
   414→          sparkData={sparklines.approval}
   415→          sparkColor="#10B981"
   416→        />
   417→        <StatCard
   418→          title="Portfolio at Risk"
   419→          value={`${summary.defaultRisk.toFixed(1)}%`}
   420→          change={-2.3}
   421→          icon={<AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />}
   422→          iconBg="bg-amber-100 dark:bg-amber-900/40"
   423→          sparkData={sparklines.risk}
   424→          sparkColor="#F59E0B"
   425→        />
   426→        <StatCard
   427→          title="Avg Credit Score"
   428→          value={summary.averageCreditScore.toFixed(1)}
   429→          change={2.1}
   430→          icon={<TrendingUp className="w-5 h-5 text-teal-600 dark:text-teal-400" />}
   431→          iconBg="bg-teal-100 dark:bg-teal-900/40"
   432→          sparkData={sparklines.score}
   433→          sparkColor="#0F766E"
   434→        />
   435→      </div>
   436→
   437→      {/* ---------- Row 2: Area Chart + Donut ---------- */}
   438→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 stagger-children">
   439→        {/* Area Chart — 2/3 width */}
   440→        <motion.div
   441→          className="lg:col-span-2"
   442→          initial={{ opacity: 0, y: 20 }}
   443→          animate={{ opacity: 1, y: 0 }}
   444→          transition={{ duration: 0.5, delay: 0.1 }}
   445→        >
   446→          <Card className="glass-card rounded-xl border border-border/50 shadow-sm">
   447→            <CardHeader className="pb-2">
   448→              <div className="flex items-center justify-between">
   449→                <CardTitle className="text-base font-semibold">Applications Over Time</CardTitle>
   450→                <Button variant="ghost" size="icon" className="h-8 w-8">
   451→                  <MoreHorizontal className="w-4 h-4" />
   452→                </Button>
   453→              </div>
   454→            </CardHeader>
   455→            <CardContent className="pt-0">
   456→              <ResponsiveContainer width="100%" height={300}>
   457→                <AreaChart
   458→                  data={applicationsOverTime}
   459→                  margin={{ top: 10, right: 10, left: -10, bottom: 0 }}
   460→                >
   461→                  <defs>
   462→                    <linearGradient id="gradApproved" x1="0" y1="0" x2="0" y2="1">
   463→                      <stop offset="5%" stopColor="#0F766E" stopOpacity={0.3} />
   464→                      <stop offset="95%" stopColor="#0F766E" stopOpacity={0.02} />
   465→                    </linearGradient>
   466→                    <linearGradient id="gradPending" x1="0" y1="0" x2="0" y2="1">
   467→                      <stop offset="5%" stopColor="#F59E0B" stopOpacity={0.3} />
   468→                      <stop offset="95%" stopColor="#F59E0B" stopOpacity={0.02} />
   469→                    </linearGradient>
   470→                    <linearGradient id="gradRejected" x1="0" y1="0" x2="0" y2="1">
   471→                      <stop offset="5%" stopColor="#EF4444" stopOpacity={0.3} />
   472→                      <stop offset="95%" stopColor="#EF4444" stopOpacity={0.02} />
   473→                    </linearGradient>
   474→                  </defs>
   475→                  <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="var(--border)" />
   476→                  <XAxis

   478→                    axisLine={false}
   479→                    tickLine={false}

   482→                  <YAxis

   486→                  />
   487→                  <Tooltip content={<CustomAreaTooltip />} />
   488→                  <Area

   490→                    dataKey="approved"
   491→                    name="Approved"
   492→                    stackId="1"

   497→                  <Area
   498→                    type="monotone"
   499→                    dataKey="pending"
   500→                    name="Pending"
   501→                    stackId="1"
   502→                    stroke="#F59E0B"
   503→                    strokeWidth={2}
   504→                    fill="url(#gradPending)"
   505→                  />
   506→                  <Area
   507→                    type="monotone"
   508→                    dataKey="rejected"
   509→                    name="Rejected"
   510→                    stackId="1"
   511→                    stroke="#EF4444"
   512→                    strokeWidth={2}

   514→                  />
   515→                </AreaChart>

   517→            </CardContent>

   521→        {/* Pie Chart — 1/3 width */}
   522→        <motion.div
   523→          initial={{ opacity: 0, y: 20 }}
   524→          animate={{ opacity: 1, y: 0 }}
   525→          transition={{ duration: 0.5, delay: 0.2 }}
   526→        >
   527→          <Card className="glass-card rounded-xl border border-border/50 shadow-sm">
   528→            <CardHeader className="pb-2">
   529→              <div className="flex items-center justify-between">
   530→                <CardTitle className="text-base font-semibold">Risk Grade Distribution</CardTitle>

   536→            <CardContent className="pt-0">
   537→              <ResponsiveContainer width="100%" height={220}>

   549→                  >
